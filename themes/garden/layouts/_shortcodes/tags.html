{{- /* Garden — label/value tag pairs. The parser is upstream's, unchanged: "tag: [value] name: [name] color: [color]", split on "tag:". THE COLOR IS PARSED AND DROPPED — same reasoning as the message shortcode: gold is the only accent, so every pair renders as one neutral label plus one filled value, the series-tag treatment. */ -}} {{- $tagsParam := .Get "tags" | default "" -}} {{- if $tagsParam -}}
{{- $tagsList := split $tagsParam "tag:" -}} {{- range $index, $element := $tagsList -}} {{- if ne $index 0 -}} {{- $tagInfo := $element | strings.TrimSpace -}} {{- if $tagInfo -}} {{- $tagParts := split $tagInfo " " -}} {{- $tagValue := "" -}} {{- $tagName := "" -}} {{- $currentKey := "tag" -}} {{- range $part := $tagParts -}} {{- if eq $part "name:" -}} {{- $currentKey = "name" -}} {{- else if eq $part "color:" -}} {{- $currentKey = "color" -}} {{- else -}} {{- if eq $currentKey "tag" -}} {{- $tagValue = strings.TrimSpace (printf "%s %s" $tagValue $part) -}} {{- else if eq $currentKey "name" -}} {{- $tagName = strings.TrimSpace (printf "%s %s" $tagName $part) -}} {{- else if eq $currentKey "color" -}} {{- $currentKey = "done" -}} {{- end -}} {{- end -}} {{- end -}} {{- if $tagValue -}} {{- with $tagName -}} {{ . }} {{- end -}} {{ $tagValue }} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
{{- end -}}
