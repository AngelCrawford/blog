{{/* Tags Shortcode Parameters: - heading: The heading text for the tags (optional) - tags: Space-separated list of tags in format "tag: [value] name: [name] color: [color]" (optional) Example usage: {{< tags heading="Genre" tags="tag: Action name: Genre color: primary tag: Adventure name: Genre color: info tag: Sci-Fi name: Genre color: danger" >}} */}} {{- $heading := .Get "heading" -}} {{- $tagsParam := .Get "tags" | default "" -}}
{{- if $tagsParam -}} {{- $tagsList := split $tagsParam "tag:" -}}
{{- range $index, $element := $tagsList -}} {{- if ne $index 0 -}} {{- $tagInfo := $element | strings.TrimSpace -}} {{- if $tagInfo -}} {{- $tagParts := split $tagInfo " " -}} {{- $tagValue := "" -}} {{- $tagName := "" -}} {{- $tagColor := "primary" -}} {{- $currentKey := "tag" -}} {{- range $partIndex, $part := $tagParts -}} {{- if eq $part "name:" -}} {{- $currentKey = "name" -}} {{- else if eq $part "color:" -}} {{- $currentKey = "color" -}} {{- else -}} {{- if eq $currentKey "tag" -}} {{- if eq $tagValue "" -}} {{- $tagValue = $part -}} {{- else -}} {{- $tagValue = printf "%s %s" $tagValue $part -}} {{- end -}} {{- else if eq $currentKey "name" -}} {{- if eq $tagName "" -}} {{- $tagName = $part -}} {{- else -}} {{- $tagName = printf "%s %s" $tagName $part -}} {{- end -}} {{- else if eq $currentKey "color" -}} {{- $tagColor = $part -}} {{- $currentKey = "done" -}} {{- end -}} {{- end -}} {{- end -}} {{- if $tagValue -}}
{{- if $tagName -}} {{ $tagName }} {{- end -}} {{ $tagValue }}
{{- end -}} {{- end -}} {{- end -}} {{- end -}}
{{- end -}}
