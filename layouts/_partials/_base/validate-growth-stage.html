{{- /* Layer 3 of the digital-garden growth_stage validation (single source: schemas/frontmatter/article.schema.json). Halts the Hugo build with errorf when .Params.growth_stage is set to a value outside the allowed enum. Missing/empty values are intentionally NOT errored here — consumers use `default "seedling" .Params.growth_stage`. Mirrors the errorf style used in layouts/_partials/card.html. */ -}} {{- $allowed := slice "seedling" "budding" "evergreen" "withered" -}} {{- $stage := .Params.growth_stage -}} {{- if $stage -}} {{- if not (in $allowed $stage) -}} {{- errorf "Invalid growth_stage %q in %s — allowed: seedling, budding, evergreen, withered" $stage .File.Path -}} {{- end -}} {{- /* Story 1.4: withered articles MUST declare withered_date so the warning banner can render. */ -}} {{- if eq $stage "withered" -}} {{- $dateVal := strings.TrimSpace (string .Params.withered_date) -}} {{- if not $dateVal -}} {{- errorf "Missing withered_date in %s — growth_stage=%q requires withered_date (YYYY-MM-DD)" .File.Path $stage -}} {{- else if not (findRE `^\d{4}-\d{2}-\d{2}$` $dateVal) -}} {{- errorf "Invalid withered_date %q in %s — must be YYYY-MM-DD (e.g. 2026-04-15)" $dateVal .File.Path -}} {{- end -}} {{- end -}} {{- end -}}
